async function generateSoal() { const apiKey = document.getElementById('apiKey').value.trim(); if (!apiKey) { alert('Mohon masukkan API Key Anda terlebih dahulu!'); return; } const btn = document.getElementById('btnGenerate'); btn.disabled = true; btn.innerHTML = '⏳ Sedang Memproses...'; const payload = { jenjang: document.getElementById('jenjang').value, kelas: document.getElementById('kelas').value, matpel: document.getElementById('matpel').value, kurikulum: document.getElementById('kurikulum').value, topik: document.getElementById('topik').value, jumlah: document.getElementById('jumlah').value, bentuk: document.getElementById('bentuk').value, level: document.getElementById('level').value }; const promptText = `Anda adalah sistem pembuat soal ujian profesional. Buatkan dokumen soal lengkap berdasarkan ketentuan berikut: - Jenjang: ${payload.jenjang} - Kelas: ${payload.kelas} - Mata Pelajaran: ${payload.matpel} - Kurikulum: ${payload.kurikulum} - Materi/Topik: ${payload.topik} - Jumlah Soal: ${payload.jumlah} - Bentuk Soal: ${payload.bentuk} - Level Kognitif: ${payload.level} Berikan output langsung dalam format HTML murni (tanpa markdown '''html) dengan struktur: 1. Header Identitas Sekolah / Ujian 2. Tabel Kisi-kisi Soal 3. Bagian Naskah Soal Lengkap 4. Tabel Kunci Jawaban & Pembahasan Lengkap 5. Lembar Pengesahan Tanda Tangan (Kepala Sekolah & Guru Matpel) Gunakan styling HTML inline yang rapi, bersih, profesional seperti dokumen ujian cetak A4 resmi di Indonesia.`; // 1. Menggunakan Header 'x-goog-api-key' resmi untuk format kunci AQ... // 2. Menggunakan endpoint 'gemini-1.5-flash' const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent`; try { const response = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json', 'x-goog-api-key': apiKey }, body: JSON.stringify({ contents: [{ parts: [{ text: promptText }] }] }) }); const data = await response.json(); if (response.ok && data.candidates && data.candidates[0].content.parts[0].text) { let htmlResult = data.candidates[0].content.parts[0].text; htmlResult = htmlResult.replace(/```html/g, '').replace(/```/g, ''); document.getElementById('placeholder').classList.add('hidden'); const paper = document.getElementById('documentPaper'); paper.innerHTML = htmlResult; paper.classList.remove('hidden'); document.getElementById('toolbar').classList.remove('hidden'); } else { console.error('API Error:', data); alert('Gagal menghasilkan soal: ' + (data.error ? data.error.message : 'Respons tidak valid dari server.')); } } catch (err) { console.error('Fetch Error:', err); alert('Terjadi kesalahan koneksi atau jaringan.'); } finally { btn.disabled = false; btn.innerHTML = '🧪 BUAT SOAL'; } }
